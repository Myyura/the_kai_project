#define OK      1
#define ERROR   0

extern int error;

int calc(char *exp[], int *xp) {
    init_stack();
    for (int i = 0; exp[i] != NULL; i++) {
        if (isdigit(exp[i][0]))
            do_num(exp[i]);
        else if (strcmp(exp[i], "+") == 0)
            do_add();
        else if (strcmp(exp[i], "-") == 0)
            do_sub();
        else if (strcmp(exp[i], "==") == 0)
            do_eq();
        else if (strcmp(exp[i], "!") == 0)
            do_not();
        else if (strcmp(exp[i], "?") == 0)
            do_cond();
        else {
            free_stack();
            return (ERROR);
        }
    }
    *xp = pop();
    free_stack();
    if (error)  /*(X)*/
        return (ERROR);
    else
        return (OK);
}

    int a;
    a = atoi(s);
    push(a);
}

void do_add() {
    int a, b;
    b = pop();
    a = pop();
    push(a + b);
}

void do_eq() {
    int a, b;
    b = pop();
    a = pop();
    push(a == b);
}

void do_not() {

    a = pop();

}

### 题目描述

用树结构表示由整数和表 1 所列运算符组成的表达式。图 1 给出了这类表达式树的示例。树的深度优先遍历方式包括前序（pre-order）、中序（in-order）和后序（post-order）遍历。

对图 1(A) 的树作中序遍历，可得到下面的中缀表达式；括号用于明确运算顺序。

    ((10 + 20) - 30)

对表达式树作后序遍历，则可得到后缀表达式（逆波兰式）。图 1(A) 对应的后缀表达式如下，其中各元素以空格分隔。

    10 20 + 30 -

#### <center>表 1：整数运算符

| 运算符 | 说明 |
| - | - |
| `+` | 接收两个参数，把二者相加并返回结果。 |
| `-` | 接收两个参数，用第一个值减去第二个值并返回结果。 |
| `==` | 接收两个参数并比较二者；相等时返回真（非 0 值），否则返回假（0）。 |
| `!` | 接收一个参数；参数为真时返回假，否则返回真。 |
| `?` | 接收三个参数并检查第一个值；第一个值为真时返回第二个值，否则返回第三个值。 |
